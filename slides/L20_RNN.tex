\begin{frame}{What goes wrong with the networks of Lectures 18--19}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Fixed input width.}\\ A feedforward net is built for exactly $d$ inputs. A review with 12 words and one with 300 words do not fit the same input layer.
      \vskip1em
      \textbf{Bag of words loses order.}\\
      \begin{tabular}{@{}ll@{}}
        ``dog bites man'' & $\{\text{bites},\text{dog},\text{man}\}$ \\
        ``man bites dog'' & $\{\text{bites},\text{dog},\text{man}\}$
      \end{tabular}\\[0.3em]
      Same input vector, very different news.
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Sliding window?}\\ Feed the last $k$ words at once.
      \begin{itemize}
        \item Nothing older than $k$ steps can matter.
        \item Weights grow with $k$: $k$ words of 300 dims each into 128 units is $k\times 38{,}400$ weights.
        \item ``not'' at position 1 needs different weights from ``not'' at position 4, so each pattern is learnt $k$ times.
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip0.8em
  \begin{block}{What we want}
    One small set of weights, applied at every position, that carries a summary of everything read so far.
  \end{block}
\end{frame}

% ------------------------------------------------------------------ section 2
\sectionslide{Part 2}{The recurrent idea}

\begin{frame}{Reading with a notebook}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      Think of how you read a long sentence.
      \begin{itemize}
        \item You see one word at a time.
        \item You keep a short running note in your head: who, what, positive or negative so far.
        \item After each word you \emph{rewrite} the note using the old note and the new word.
        \item The note has a fixed size, however long the sentence.
      \end{itemize}
      \vskip0.6em
      The note is the \alert{hidden state} $\bm h_t$.
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}[node distance=1.0cm]
        \node[io] (x) {$x_t$};
        \node[cell,above=0.8cm of x] (c) {update};
        \node[io,above=0.8cm of c] (y) {$y_t$};
        \draw[arr] (x) -- (c); \draw[arr] (c) -- (y);
        \node[draw=inksoft,dashed,rounded corners=2pt,right=1.3cm of c,font=\small,align=center] (n) {note\\$h_{t-1}$};
        \draw[rec,visible on=<2->] (n.north) to[out=110,in=40] node[above,font=\scriptsize,text=brick]{read old note} (c.north east);
        \draw[rec,visible on=<3->] (c.south east) to[out=-40,in=-110] node[below,font=\scriptsize,text=brick]{write new note $h_t$} (n.south);
        \node[below=0.15cm of x,font=\scriptsize,color=inksoft] {the word you just read};
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\analogyframe{The hidden state is a cricket score you keep in your head}
{You are listening to a match on the radio. You cannot see the ground. The commentator describes one ball at a time, and after every ball you update a small scoreboard in your head: runs, wickets, overs, who is batting. You never replay the whole match; the scoreboard is enough to follow it.}
{one ball described on the radio & \maps & the input $\bm x_t$ \pause\\
 the scoreboard in your head & \maps & the hidden state $\bm h_t$ \pause\\
 it stays the same size, at ball 5 or ball 250 & \maps & $\bm h_t$ has $n$ numbers, whatever the length \pause\\
 your habit for updating the score & \maps & $W_{xh}$, $W_{hh}$ and $\tanh$ \pause\\
 ``we will win this'' & \maps & the output $\hat{\bm y}_t$ \\}

\begin{frame}{The update rule (Elman, 1990)}
  \vskip0.4em
  \[
    \bm h_t \;=\; \tanh\!\big(\,\underbrace{W_{xh}\,\bm x_t}_{\text{\color{navy}what the new input says}}
      \;+\; \underbrace{W_{hh}\,\bm h_{t-1}}_{\text{\color{brick}what the old note says}}
      \;+\; \bm b_h\big)
    \qquad\qquad
    \bm{\hat y}_t \;=\; \operatorname{softmax}\!\big(W_{hy}\,\bm h_t + \bm b_y\big)
  \]
  \vskip1.2em
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \textbf{Sizes} (input $d$, hidden $n$, output $k$)\\[0.2em]
      \begin{tabular}{@{}ll@{}}
        $W_{xh}$ & $n\times d$ \\
        $W_{hh}$ & $n\times n$ \quad \alert{the recurrence} \\
        $W_{hy}$ & $k\times n$ \\
        $\bm h_0$ & usually all zeros
      \end{tabular}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Why $\tanh$?}\\
      It keeps every entry of $\bm h_t$ between $-1$ and $1$, so the note cannot blow up however many steps we run, and it is zero-centred, so positive and negative evidence can cancel.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{One set of weights for any length}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Parameter count}
      \[
        \underbrace{n d}_{W_{xh}} + \underbrace{n^2}_{W_{hh}} + \underbrace{n}_{\bm b_h} + \underbrace{k n + k}_{\text{output}}
      \]
      Example: $d=3$, $n=4$, $k=2$:
      \[ 12 + 16 + 4 + 8 + 2 = 42 \]
      Feed it 10 words or 10{,}000 words: still 42 numbers to learn.
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{Why sharing helps}
      \begin{itemize}
        \item A pattern learnt at position 3 (``not good'') works at position 30.
        \item Fewer weights means less data needed, the same argument as weight sharing in CNNs, but along time instead of space.
        \item The price: the same $W_{hh}$ is multiplied in again and again. Hold on to that; it comes back in Part 4.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\analogyframe{Parameter sharing is one marking scheme for every script}
{An examiner values 300 answer scripts with a single marking scheme. The scheme does not get longer because there are more scripts, and an answer earns the same marks whether it is on page 2 or page 12.}
{the marking scheme & \maps & the weights $W_{xh}$, $W_{hh}$ \pause\\
 the same scheme for script 1 and script 300 & \maps & the same weights at step 1 and step 300 \pause\\
 a good point is rewarded wherever it appears & \maps & ``not good'' is understood at any position \\}

\begin{frame}{A forward pass you can do by hand}
  \small
  Scalar RNN, no bias: \quad $h_t=\tanh(0.5\,x_t + 0.8\,h_{t-1})$, \quad $h_0=0$, \quad inputs $x = (1,\;0,\;2)$
  \vskip0.8em
  \renewcommand{\arraystretch}{1.25}
  \begin{tabular}{@{}c c l l c@{}}
    \toprule
    $t$ & $x_t$ & pre-activation $z_t = 0.5x_t + 0.8h_{t-1}$ & & $h_t=\tanh z_t$ \\ \midrule
    1 & 1 & $0.5(1) + 0.8(0)$ & $=0.500$ & \uncover<2->{$0.462$} \\
    2 & 0 & $0.5(0) + 0.8(0.462)$ & $=0.370$ & \uncover<3->{$0.354$} \\
    3 & 2 & $0.5(2) + 0.8(0.354)$ & $=1.283$ & \uncover<4->{$0.857$} \\ \bottomrule
  \end{tabular}
  \vskip1em
  \uncover<4->{
  \begin{itemize}
    \item At $t=2$ the input is $0$, yet $h_2\neq 0$: the network still remembers $x_1$, shrunk by $0.8$.
    \item With no new input, the memory of $x_1$ fades by roughly a factor $w_{hh}=0.8$ per step. Keep that number in mind for the game.
  \end{itemize}}
\end{frame}

\sectionslide{Part 4}{Training: backpropagation through time}

\begin{frame}{Unroll, then backpropagate as usual}
  \begin{columns}[c]
    \begin{column}{0.55\textwidth}
      \centering
      \begin{tikzpicture}[x=1.55cm]
        \foreach \i in {1,2,3,4} {
          \node[io,font=\scriptsize] (x\i) at (\i,0) {$x_\i$};
          \node[cell,minimum width=0.85cm] (h\i) at (\i,1.3) {$h_\i$};
          \node[io,font=\scriptsize] (L\i) at (\i,2.6) {$L_\i$};
          \draw[arr] (x\i)--(h\i); \draw[arr] (h\i)--(L\i);
        }
        \foreach \i [evaluate=\i as \j using int(\i+1)] in {1,2,3} { \draw[arr] (h\i)--(h\j); }
        \foreach \i [evaluate=\i as \j using int(\i+1)] in {1,2,3} {
          \draw[rec,dashed,visible on=<2->] ([yshift=-0.25cm]h\j.south west) to[bend left=25] ([yshift=-0.25cm]h\i.south east);
        }
        \node[font=\scriptsize,text=brick,visible on=<2->] at (2.5,-0.75) {dashed: gradients flow back in time};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{itemize}
        \item Total loss is the sum over steps:\\ $L = \sum_t L_t(\hat y_t, y_t)$.
        \item The unrolled graph is just a deep feedforward net with tied weights.
        \item Run ordinary backprop (Lecture 19) on it. Gradients for the shared $W_{hh}$ are \emph{added up} over all steps.
        \item This is \alert{backpropagation through time} (BPTT).
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip0.2em
  \margin{For long sequences we cut the unrolled graph every $K$ steps (truncated BPTT) to save memory.}
\end{frame}

\analogyframe{Backpropagation through time is a post-match review}
{The team lost by 12 runs. In the review the coach walks back over by over, asking how much each over contributed to the loss. A bowler who bowled five overs collects blame from all five.}
{the final result & \maps & the loss $L$ \pause\\
 walking back over by over & \maps & backpropagating step by step \pause\\
 one bowler, many overs, blame added up & \maps & one $W_{hh}$, many steps, gradients summed \pause\\
 his next-season training plan & \maps & the weight update \\}

\begin{frame}{Why the signal from far back gets lost}
  To reach step $t-k$, the gradient is multiplied by one Jacobian per step:
  \[
    \frac{\partial \bm h_t}{\partial \bm h_{t-k}} \;=\; \prod_{j=t-k+1}^{t} \operatorname{diag}\!\big(1-\bm h_j^2\big)\,W_{hh}
  \]
  \vskip0.3em
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Scalar intuition.} Drop the $\tanh$ factor and use one number $w$:
      \vskip0.3em
      \begin{tabular}{@{}lr@{}}
        $0.9^{50}$  & $0.005$ \\
        $1.0^{50}$  & $1$ \\
        $1.1^{50}$  & $117$ \\
      \end{tabular}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item $|w|<1$: the gradient \alert{vanishes}. Early words get almost no say in learning.
        \item $|w|>1$: the gradient \alert{explodes}. One bad batch throws the weights to \texttt{NaN}.
        \item The $\tanh$ factor $1-h^2\le 1$ only ever makes it smaller.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\analogyframe{Vanishing and exploding gradients are compound interest}
{Pass a whispered message down a row of 50 students and little of it survives. Put money in a deposit that loses 10\% a year and after 50 years you hold half a percent of it; one that gains 10\% a year multiplies 117 times. Training signals that travel back through time behave the same way.}
{a whisper passed along a long row & \maps & the error signal passed back step by step \pause\\
 losing 10\% a year: $0.9^{50}=0.005$ & \maps & vanishing gradient: early words are never corrected \pause\\
 gaining 10\% a year: $1.1^{50}=117$ & \maps & exploding gradient: weights jump to \texttt{NaN} \pause\\
 only a rate of exactly 0\% keeps the sum steady & \maps & only $|w_{hh}|\approx 1$ keeps the gradient alive \\}

      \small
      A $\tanh$ RNN with 64 hidden units, random inputs, $W_{hh}$ an orthogonal matrix scaled to spectral radius $\rho$. Gradient norms computed exactly by automatic differentiation.
      \vskip0.6em
      \begin{itemize}
        \item With $\rho=0.7$ the gradient is below $10^{-3}$ after 10 steps.
        \item Even $\rho=1.3$ cannot hold it up: once $\tanh$ saturates, $1-h^2$ drags it down.
      \end{itemize}

\begin{frame}{What practitioners do about it}
  \begin{tabular}{@{}>{\raggedright}p{3.4cm}>{\raggedright\arraybackslash}p{10.2cm}@{}}
    \toprule
    \textbf{Problem} & \textbf{Fix} \\ \midrule
    Exploding gradients & \textbf{Gradient clipping}: if $\|g\|>\tau$, rescale $g \leftarrow \tau\, g/\|g\|$. Cheap and used almost everywhere (Pascanu et al., 2013). \\[0.3em]
    Vanishing gradients & Careful initialisation (e.g.\ orthogonal $W_{hh}$), ReLU-type activations, skip connections in time. These help a little. \\[0.3em]
    & \textbf{Gated cells} (LSTM, GRU) that let the memory pass through by \emph{addition} instead of repeated multiplication. These help a lot, and they are the next lecture. \\[0.3em]
    Long sequences, memory & Truncated BPTT: backpropagate through the last $K$ steps only. \\ \bottomrule
  \end{tabular}
\end{frame}

\analogyframe{Gradient clipping is a speed governor}
{A KSRTC bus has a governor: however hard the driver presses, the speed never goes above the limit. The bus still goes where it was going, just not dangerously fast.}
{the accelerator pressed too hard & \maps & a huge gradient from one bad batch \pause\\
 the governor's speed limit & \maps & the threshold $\tau$ \pause\\
 same direction, safe speed & \maps & $g \leftarrow \tau\, g/\|g\|$: same direction, capped size \\}

\begin{frame}[fragile]{The whole of BPTT in twelve lines of NumPy}
\begin{lstlisting}
for t in range(len(inputs)):                                   # forward
    hs[t] = np.tanh(Wxh @ xs[t] + Whh @ hs[t-1] + bh)          # the update rule
    ps[t] = softmax(Why @ hs[t] + by)                          # next-character probabilities
    loss -= np.log(ps[t][targets[t]])                          # cross-entropy

dh_next = np.zeros((H, 1))
for t in reversed(range(len(inputs))):                         # backward, through time
    dy = ps[t].copy(); dy[targets[t]] -= 1                     # softmax + cross-entropy gradient
    dWhy += dy @ hs[t].T
    dh   = Why.T @ dy + dh_next                                # from now + from the future
    draw = (1 - hs[t]**2) * dh                                 # back through tanh
    dWxh += draw @ xs[t].T;  dWhh += draw @ hs[t-1].T          # shared weights: sums
    dh_next = Whh.T @ draw                                     # one step further back
\end{lstlisting}
  \vskip0.4em
  \small Line 13 is where vanishing happens: \texttt{dh\_next} is multiplied by $W_{hh}^\top$ and by $(1-h^2)$ at every step.\\
  \margin{Full script (about 100 lines, NumPy only): \texttt{case\_study/char\_rnn\_places.py} in the course repository.}
\end{frame}

\begin{frame}{What the game was teaching}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Memory}
      \begin{itemize}
        \item $0<w_{hh}<1$: a pulse fades geometrically.\\ Small $w_{hh}$ forgets in two or three steps.
        \item $w_{hh}$ close to 1: long memory, but every new input piles on top and the state saturates at $\pm1$.
        \item $w_{hh}<0$: the state flips sign each step.
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{Learning}
      \begin{itemize}
        \item The gradient survived 40 steps only for $|w_{hh}|$ between about $0.944$ and $1.059$.
        \item Training has to find and stay inside that thin band, for every unit, while also solving the task.
        \item That is too fragile. Next session: a cell whose memory path is built to have a factor near 1.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Exit ticket (worksheet Part D)}
  \yourturn{5 min}
  \begin{enumerate}
    \item An RNN with $d=50$ inputs and $n=64$ hidden units (no output layer) has how many parameters?\\
      (a) 3{,}264 \quad (b) 7{,}360 \quad (c) 7{,}296 \quad (d) depends on the sequence length
    \item In BPTT the gradient for $W_{hh}$ is
      (a) taken from the last step only \quad (b) averaged over epochs \quad (c) summed over all time steps \quad (d) zero for $t=1$
    \item Gradient clipping is a fix for \emph{(ask aloud; not on the sheet)}\\
      (a) vanishing gradients \quad (b) exploding gradients \quad (c) overfitting \quad (d) slow inference
  \end{enumerate}

\begin{frame}{To take away}
  \begin{enumerate}
    \item An RNN carries a fixed-size hidden state and updates it with the same weights at every step: $\bm h_t=\tanh(W_{xh}\bm x_t+W_{hh}\bm h_{t-1}+\bm b)$.
    \item Where you feed inputs and read outputs gives five shapes: one-to-many, many-to-one, aligned, encoder--decoder.
    \item Training is ordinary backpropagation on the unrolled graph, with gradients summed over time.
    \item The gradient is multiplied by $W_{hh}$ and $\tanh'$ once per step, so it vanishes or explodes over long gaps.
    \item Clipping handles explosion. Vanishing needs a different cell.
  \end{enumerate}
  \vskip1em
  \textbf{Next session (Lecture 21):} LSTM and GRU, cells with gates that decide what to forget, what to write and what to show; bidirectional RNNs; a case study that compares all four.
\end{frame}
