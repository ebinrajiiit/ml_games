\begin{frame}{How did you learn to ride a bicycle?}
  \vskip0.6em
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      Nobody gave you a dataset of\\[0.3em]
      \quad (angle of lean, speed) $\;\rightarrow\;$ correct handle turn.
      \vskip0.8em
      You tried something, wobbled, fell, tried again. Staying up felt good; falling hurt. After a few days your body had a rule for what to do in every situation.
    \end{column}
    \begin{column}{0.46\textwidth}
      \uncover<2->{
      That is the whole idea of reinforcement learning:
      \begin{itemize}
        \item no correct answer is given for each situation;
        \item you act, and the world answers with a \emph{consequence};
        \item consequences can arrive late (you fall three seconds after the bad turn);
        \item what you do changes what you see next.
      \end{itemize}}
    \end{column}
  \end{columns}
\end{frame}

        \draw[arr,visible on=<3->] ([yshift=-0.25cm]en.west) -- (-2.9,-0.25) -- node[left,font=\small]{state $s_{t+1}$} (-2.9,2.85) -- ([yshift=0.25cm]ag.west);
        \draw[rec,visible on=<4->] ([yshift=0.25cm]en.west) -- (-2.2,0.25) -- node[right,font=\small,text=brick]{reward $r_{t+1}$} (-2.2,2.35) -- ([yshift=-0.25cm]ag.west);
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.48\textwidth}
      At each time step $t = 0,1,2,\ldots$
      \begin{enumerate}[<+(1)->]
        \item the agent observes state $s_t$;
        \item it picks action $a_t$;
        \item the environment moves to $s_{t+1}$ and hands back a reward $r_{t+1}$.
      \end{enumerate}
      \vskip0.6em
      This produces a trajectory
      \[ s_0,\,a_0,\,r_1,\,s_1,\,a_1,\,r_2,\,s_2,\,\ldots \]
      Note the parallel with Lecture 20: a sequence, processed one step at a time.
    \end{column}
  \end{columns}
\end{frame}

    \pause

  \end{tabular}
  \vskip0.6em
  \small The agent's goal: find the policy $\pi$ that makes the expected return as large as possible.

\begin{frame}{The return, and why we discount}
  \[
    G_t \;=\; r_{t+1} + \gamma\, r_{t+2} + \gamma^2 r_{t+3} + \cdots \;=\; \sum_{k=0}^{\infty} \gamma^k\, r_{t+k+1}
  \]
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Worked example}, $\gamma = 0.9$\\[0.3em]
      Rewards along an episode: $-1,\;-1,\;-1,\;+10$
      \vskip0.3em
      \renewcommand{\arraystretch}{1.2}
      \begin{tabular}{@{}l@{}}
        $G_0 = -1 + 0.9(-1) + 0.81(-1) + 0.729(10)$ \\
        \uncover<2->{$\phantom{G_0} = -1 - 0.9 - 0.81 + 7.29$} \\
        \uncover<3->{$\phantom{G_0} = \alert{4.58}$}
      \end{tabular}
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{Why $\gamma<1$?}
      \begin{itemize}
        \item In a task that never ends, an undiscounted sum can be infinite.
        \item A reward now is more certain than one later.
        \item It makes the agent prefer the \emph{shorter} of two routes to the same prize.
      \end{itemize}
      \vskip0.3em
      $\gamma=0$: only the next reward matters (short-sighted).\\
      $\gamma\to1$: all future rewards count almost equally.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The Markov property: is your state enough?}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      A state is \textbf{Markov} if the future depends only on it, not on how you got there:
      \[ P(s_{t+1}, r_{t+1}\mid s_t, a_t) = P(s_{t+1}, r_{t+1}\mid s_0,a_0,\ldots,s_t,a_t) \]
      With Markov states and actions, the task is a \textbf{Markov decision process} (MDP):
      \[ (\mathcal S,\ \mathcal A,\ P,\ R,\ \gamma) \]
    \end{column}
    \begin{column}{0.45\textwidth}
      \textbf{Quick check}
      \begin{itemize}
        \item Chess: the board position (plus whose turn, castling rights). Markov.
        \item Pong from \emph{one} screenshot: you see the ball, not where it is going. Not Markov.\\ Fix: stack the last 4 frames (this is what the Atari DQN did).
        \item A patient's vitals right now, for a dosing agent: probably not; add recent history.
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip0.6em
  \margin{Link to Lectures 20--21: when the raw observation is not Markov, an RNN or LSTM can summarise the history into a state.}
\end{frame}

\sectionslide{Part 2}{Formulating a task: five questions}

\begin{frame}{A recipe for turning a problem into an RL task}
  \begin{enumerate}
    \item \textbf{Who is the agent, and what is the environment?} Draw the boundary. Anything the agent cannot change directly is environment.
    \item \textbf{State space $\mathcal S$.} What must the agent see to choose well? Is it Markov? Discrete or continuous? How big?
    \item \textbf{Action space $\mathcal A$.} Discrete (left/right/stay) or continuous (steering angle in $[-30^\circ,30^\circ]$)? Are some actions illegal in some states?
    \item \textbf{Reward $R$.} What number, when? Reward the \emph{outcome you want}, not the method you guess will get there.
    \item \textbf{Episodes.} Is there a natural end (episodic: a game, a delivery) or does it run for ever (continuing: a thermostat)? What is $\gamma$? Is there a time limit?
  \end{enumerate}
  \vskip0.6em
  \small Sutton and Barto, Chapter 3, is the reference for this lecture.
\end{frame}

\begin{frame}{When the reward says the wrong thing}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      An agent maximises exactly the number you give it, and nothing else.
      \vskip0.5em
      \textbf{A real case.} OpenAI trained an agent on the boat-racing game \emph{CoastRunners}, rewarding the in-game score. The agent found a lagoon where three targets kept respawning, and circled there for ever, on fire, crashing into boats, never finishing the race. It scored higher than human players (Clark and Amodei, 2016).
      \vskip0.5em
      This is called \textbf{reward hacking} or \textbf{specification gaming}.
    \end{column}
    \begin{column}{0.45\textwidth}
      \textbf{Common mistakes}
      \begin{itemize}
        \item Rewarding activity (``+1 per step working'') instead of results.
        \item Rewarding a sub-goal so richly that it beats the goal.
        \item Forgetting a cost (energy, wear, safety) the designer cared about but never wrote down.
        \item Equal rewards for slow and fast success, with $\gamma=1$.
      \end{itemize}
      \vskip0.4em
      You will make all four on purpose in today's game.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Defining it in code: the Gymnasium interface}
\begin{lstlisting}
class CampusCourier(gym.Env):
    def __init__(self, step_reward=-1.0, puddle_reward=-10.0, goal_reward=10.0, max_steps=30):
        self.observation_space = spaces.Discrete(25)       # S: 25 cells
        self.action_space      = spaces.Discrete(4)        # A: up, right, down, left
        ...
    def reset(self, seed=None, options=None):
        super().reset(seed=seed)
        self.pos, self.t = (4, 2), 0                       # start at the hostel
        return self._obs(), {}

    def step(self, action):
        dr, dc = self.MOVES[action]
        r, c = self.pos[0] + dr, self.pos[1] + dc
        if 0 <= r < 5 and 0 <= c < 5: self.pos = (r, c)    # walls: stay put
        self.t += 1
        terminated = self.pos == self.GOAL                 # a real end
        truncated  = self.t >= self.max_steps              # an artificial time limit
        reward = (self.goal_reward if terminated else
                  self.puddle_reward if self.pos in self.PUDDLE else self.step_reward)
        return self._obs(), reward, terminated, truncated, {}
\end{lstlisting}
  \small Every RL library expects exactly these two methods. \margin{Full file, with a test: \texttt{case\_study/campus\_courier\_env.py}. It passes Gymnasium's \texttt{check\_env}.}
\end{frame}

\begin{frame}{What the game was teaching}
  \begin{itemize}
    \item \textbf{Challenge 1.} A positive reward per step makes staying alive the goal. Any step reward $\le 0$ fixed it. So did a goal reward well above 20, the value of loitering for ever at $+1$ a step.
    \item \textbf{Challenge 2.} The robot gets wet unless the puddle penalty outweighs the extra moves. With step $-1$ and $\gamma=0.95$ the switch is at a puddle reward of about $-6.02$: solve $6.67+0.95\,r_p = 0.95$.
    \item \textbf{Challenge 3.} With no step cost and $\gamma=1$, every route to the library is worth the same $+10$, so the robot has no reason to arrive. Any $\gamma<1$ makes sooner better than later.
  \end{itemize}
  \vskip0.6em
  \begin{block}{The point}
    The algorithm was the same in every case. Only the formulation changed, and it decided everything.
  \end{block}
\end{frame}

\begin{frame}{To take away}
  \begin{enumerate}
    \item RL learns from rewards for actions, not from labelled answers; the feedback is delayed and the data depends on the learner's own choices.
    \item Formulating a task means fixing the agent, the state space, the action space, the reward, and the episode structure with $\gamma$.
    \item The return $G_t=\sum_k \gamma^k r_{t+k+1}$ is what the agent maximises; $\gamma$ sets how far ahead it looks.
    \item Check the state is Markov; if not, add history.
    \item The agent optimises exactly the reward you wrote. Test it with a random policy and with hand-written routes before training anything.
  \end{enumerate}
